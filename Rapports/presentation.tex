\section{Contexte et problématique}

\begin{frame}{Contexte : la cybersécurité au Tchad}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \begin{itemize}
        \item<1-> \textbf{2,74 M d'internautes} et 14,5 M de connexions
              mobiles au Tchad en 2025
        \item<2-> La digitalisation avance vite, mais les outils de
              cybersécurité restent \textbf{quasi inexistants}
        \item<3-> Peu de SOC nationaux, peu d'experts locaux
      \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
      \onslide<2->{
      \begin{block}{Exemple réel}
        \textbf{10 jan. 2024} : attaque DDoS coupe
        \textbf{tout Internet} au Tchad pendant plusieurs heures.
      \end{block}
      }
    \end{column}
  \end{columns}
  \vspace{0.8em}
  \onslide<3->{\centering\alert{La cybersécurité n'est plus un luxe : c'est une nécessité.}}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Le problème}
  \onslide<1->{
  Les outils classiques (Snort, Suricata\ldots) fonctionnent sur des \textbf{règles fixes} :
  \begin{center}
  \begin{tikzpicture}
    \node[draw=csblue, rounded corners, fill=csblue!8,
          minimum width=9cm, minimum height=0.9cm, align=center, font=\small]
      {``Si tu vois EXACTEMENT ce motif $\rightarrow$ alerte''};
  \end{tikzpicture}
  \end{center}
  }

  \vspace{0.5em}
  \begin{itemize}
    \item<2-> \textcolor{csviolet}{\textbf{Limite 1}} : une attaque \textbf{nouvelle} passe inaperçue
    \item<3-> \textcolor{csviolet}{\textbf{Limite 2}} : les alertes sont \textbf{incompréhensibles}
              pour un analyste débutant
  \end{itemize}

  \vspace{0.6em}
  \onslide<4->{
  \begin{block}{Notre question centrale}
    Comment détecter des attaques \textbf{même inconnues} et les
    \textbf{expliquer en langage simple}, en gardant l'\textbf{humain} dans la boucle ?
  \end{block}
  }
\end{frame}

\section{Bilan et perspectives}

\begin{frame}{Ce qui fonctionne}
  \begin{itemize}
    \item<1-> Détection ML en \textbf{moins de 50 ms}
    \item<2-> RAG sur \textbf{697 techniques} MITRE ATT\&CK
    \item<3-> Explication LLM en \textbf{30--90 secondes}
    \item<4-> Blocage avec \textbf{validation humaine} obligatoire
    \item<5-> Interface temps réel via SSE (sans rechargement)
    \item<6-> Tests automatisés backend + frontend
  \end{itemize}
\end{frame}
