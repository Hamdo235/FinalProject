\section{Contexte et problématique}

\begin{frame}{Contexte : la cybersécurité au Tchad}
  \begin{columns}[T]
    \begin{column}{0.55\textwidth}
      \begin{itemize}
        \item En 2025, le Tchad compte \textbf{2,74 millions d'internautes} et
              \textbf{14,5 millions de connexions mobiles}
        \item La digitalisation avance vite, mais les outils de cybersécurité
              restent \textbf{limités}
        \item Peu de SOC nationaux, peu d'experts locaux, dépendance à l'étranger
      \end{itemize}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{block}{Un exemple concret}
        Le \textbf{10 janvier 2024}, une attaque DDoS a coupé
        \textbf{tout Internet} au Tchad pendant plusieurs heures.
      \end{block}
    \end{column}
  \end{columns}
  \vspace{1em}
  \centering
  \alert{La cybersécurité n'est plus un luxe : c'est une nécessité.}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Le problème}
  Les outils classiques (Snort, Suricata...) fonctionnent avec des
  \textbf{règles fixes} :

  \vspace{0.5em}
  \begin{center}
  \begin{tikzpicture}
    \node[draw=csblue, rounded corners, fill=csblue!8, minimum width=10cm, minimum height=1.1cm, align=center]
      (regle) {``Si tu vois EXACTEMENT ce motif $\rightarrow$ alerte''};
  \end{tikzpicture}
  \end{center}

  \vspace{0.8em}
  \begin{itemize}
    \item \textcolor{csviolet}{\textbf{Limite 1}} : une attaque \textbf{nouvelle}
          (jamais vue) passe inaperçue
    \item \textcolor{csviolet}{\textbf{Limite 2}} : les alertes sont
          \textbf{trop techniques} pour un analyste débutant
          \\ \small (ex : \texttt{score=0.87, rule=ml\_iforest\_anomaly} --- et maintenant ?)
  \end{itemize}

  \vspace{0.8em}
  \begin{block}{Notre question centrale}
    Comment détecter des attaques \textbf{même inconnues}, les
    \textbf{expliquer en langage simple}, tout en gardant un
    \textbf{humain} dans la décision finale --- sur une machine
    \textbf{sans GPU} ?
  \end{block}
\end{frame}

\begin{frame}{L'idée : 3 cerveaux qui travaillent ensemble}
  \begin{center}
  \begin{tikzpicture}[

  \vspace{1.2em}
  \begin{exampleblock}{Principe central : Human-in-the-Loop}
    L'IA \textbf{analyse, explique, recommande} --- mais c'est toujours
    \textbf{l'analyste humain} qui décide de bloquer ou non.
  \end{exampleblock}
\end{frame}

% ------------------------------------------------------------
\begin{frame}{Objectif général}
  \begin{block}{}
    Concevoir et réaliser une plateforme qui combine \textbf{détection
    d'anomalies} (Machine Learning), \textbf{explication contextuelle}
    (RAG + MITRE ATT\&CK + LLM local), et \textbf{neutralisation adaptive}
    des menaces validée par un humain --- pour aider un analyste SOC à
    comprendre et réagir plus vite.
  \end{block}

  \vspace{1em}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Contraintes respectées :}
      \begin{itemize}
        \item Fonctionne \textbf{sans GPU} (CPU + 16 Go RAM)
        \item \textbf{100\% local} --- aucune donnée envoyée au cloud
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \textbf{Technologies clés :}
      \begin{itemize}
        \item Backend : Python / FastAPI
        \item Frontend : Next.js / React
        \item Base : SQLite + ChromaDB
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Comment ça marche ? Exemple concret}
  \small
  \textit{Une connexion SSH échoue 3 fois en 1h, à 3h du matin, depuis une IP inconnue.}

  \vspace{0.5em}
  \begin{center}
  \begin{tikzpicture}[
      stepbox/.style={draw, rounded corners, minimum width=2.2cm, minimum height=1.5cm,
                    align=center, font=\tiny, drop shadow},
      arr/.style={-{Latex}, thick, csgray}
    ]
    \node[stepbox, fill=csblue!12, draw=csblue] (a) at (0,0) {Log reçu\\\texttt{/ingest/logs}};
    \node[stepbox, fill=csteal!15, draw=csteal] (b) at (2.6,0) {Score ML\\$0,87$ (high)};
    \node[stepbox, fill=csviolet!12, draw=csviolet] (c) at (5.2,0) {Défense propose\\``Bloquer ?''};
    \node[stepbox, fill=csteal!15, draw=csteal] (d) at (7.8,0) {RAG trouve\\T1110.001};
    \node[stepbox, fill=csblue!12, draw=csblue] (e) at (10.4,0) {LLM explique\\en français};

    \draw[arr] (a) -- (b);
    \draw[arr] (b) -- (c);
    \draw[arr] (c) -- (d);
    \draw[arr] (d) -- (e);
  \end{tikzpicture}
  \end{center}

  \vspace{1em}
  \begin{block}{Score de confiance final}
    \centering
    confiance $= 0{,}4 \times \text{score}_{ML} + 0{,}3 \times \text{score}_{RAG} + 0{,}3 \times \text{score}_{LLM}$

    \small $0{,}4{\times}0{,}87 + 0{,}3{\times}0{,}82 + 0{,}3{\times}0{,}89 = \mathbf{0{,}861}$
    $\rightarrow$ \textbf{86\% de confiance}
  \end{block}

  \vspace{0.5em}
  \centering
  L'analyste reçoit une notification, \textbf{décide}, et l'IP est bloquée
  pendant 30 min seulement si \textbf{lui} l'approuve.
\end{frame}

\begin{frame}{Limites et perspectives}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Limites actuelles}
      \begin{itemize}
        \item Authentification simplifiée (côté frontend)
        \item Blocage iptables \textbf{affiché}, pas exécuté
              automatiquement

      \begin{itemize}
        \item Couvrir plus de types d'attaques (DDoS, injection...)
        \item Exécution automatique sécurisée des blocages
        \item Authentification renforcée (JWT, rôles)
        \item Adaptation au contexte tchadien (ANSICE)
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
